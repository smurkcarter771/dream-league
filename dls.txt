    <!-- JavaScript Application Logic -->
    <script>
        // Game State & LocalStorage
        let gameState = JSON.parse(localStorage.getItem('dls19_save')) || {
            coins: 1500,
            teamName: "Dream FC",
            stadiumName: "Global Stadium",
            stadiumCapacity: 15000,
            primaryColor: "bg-blue-600",
            stats: { wins: 0, draws: 0, losses: 0 },
            squad: [
                { name: "G. Buffon", pos: "GK", ovr: 88, price: 0 },
                { name: "Sergio Ramos", pos: "DEF", ovr: 90, price: 0 },
                { name: "G. Chiellini", pos: "DEF", ovr: 87, price: 0 },
                { name: "L. Modric", pos: "MID", ovr: 89, price: 0 },
                { name: "T. Kroos", pos: "MID", ovr: 88, price: 0 },
                { name: "C. Ronaldo", pos: "FWD", ovr: 94, price: 0 },
                { name: "L. Messi", pos: "FWD", ovr: 95, price: 0 }
            ],
            market: [
                { id: 1, name: "Neymar Jr", pos: "FWD", ovr: 92, price: 1200 },
                { id: 2, name: "K. De Bruyne", pos: "MID", ovr: 91, price: 1100 },
                { id: 3, name: "V. Van Dijk", pos: "DEF", ovr: 90, price: 1000 },
                { id: 4, name: "M. Salah", pos: "FWD", ovr: 90, price: 950 },
                { id: 5, name: "P. Pogba", pos: "MID", ovr: 87, price: 750 },
                { id: 6, name: "N. Kanté", pos: "MID", ovr: 89, price: 900 }
            ]
        };

        function saveGame() {
            localStorage.setItem('dls19_save', JSON.stringify(gameState));
            updateUI();
        }

        function updateUI() {
            document.getElementById('coinDisplay').innerText = gameState.coins.toLocaleString();
            document.getElementById('headerTeamName').innerText = gameState.teamName;
            document.getElementById('clubNameDisplay').innerText = gameState.teamName;
            document.getElementById('stadiumNameDisplay').innerText = `${gameState.stadiumName} (Cap: ${gameState.stadiumCapacity.toLocaleString()})`;
            document.getElementById('currentStadiumTier').innerText = `${gameState.stadiumName} (${gameState.stadiumCapacity.toLocaleString()} Cap)`;
            
            document.getElementById('statWins').innerText = gameState.stats.wins;
            document.getElementById('statDraws').innerText = gameState.stats.draws;
            document.getElementById('statLosses').innerText = gameState.stats.losses;

            // Calculate Team OVR
            if (gameState.squad.length > 0) {
                let totalOvr = gameState.squad.reduce((sum, p) => sum + p.ovr, 0);
                document.getElementById('ovrDisplay').innerText = Math.round(totalOvr / gameState.squad.length);
            }

            renderSquad();
            renderMarket();
        }

        function switchTab(tabId) {
            ['hub', 'market', 'match', 'stadium'].forEach(id => {
                document.getElementById(`tab-${id}`).classList.add('hidden');
                document.getElementById(`nav-${id}`).className = "tab-btn px-4 py-2 rounded-lg font-bold text-sm tracking-wide transition text-slate-400 hover:text-white hover:bg-slate-800";
            });
            document.getElementById(`tab-${tabId}`).classList.remove('hidden');
            document.getElementById(`nav-${tabId}`).className = "tab-btn px-4 py-2 rounded-lg font-bold text-sm tracking-wide transition bg-blue-600 text-white shadow";
        }

        function renderSquad() {
            const grid = document.getElementById('squadGrid');
            grid.innerHTML = '';
            gameState.squad.forEach(player => {
                grid.innerHTML += `
                    <div class="bg-slate-900 border border-slate-800 p-4 rounded-xl flex justify-between items-center shadow">
                        <div>
                            <span class="text-xs font-bold px-2 py-0.5 rounded bg-slate-800 text-slate-300 mr-2">${player.pos}</span>
                            <span class="font-bold text-white text-base">${player.name}</span>
                        </div>
                        <div class="bg-slate-950 px-3 py-1 rounded border border-slate-800 text-blue-400 font-bold text-sm">
                            ${player.ovr} OVR
                        </div>
                    </div>
                `;
            });
        }

        function renderMarket() {
            const grid = document.getElementById('marketGrid');
            grid.innerHTML = '';
            if (gameState.market.length === 0) {
                grid.innerHTML = `<p class="text-slate-500 col-span-full text-center py-8">Transfer market is currently empty. Click refresh to scout new players!</p>`;
                return;
            }
            gameState.market.forEach(player => {
                grid.innerHTML += `
                    <div class="bg-slate-900 border border-slate-800 p-5 rounded-xl flex flex-col justify-between shadow space-y-4">
                        <div class="flex justify-between items-start">
                            <div>
                                <span class="text-xs font-bold px-2 py-0.5 rounded bg-slate-800 text-slate-300">${player.pos}</span>
                                <h3 class="font-bold text-lg text-white mt-1">${player.name}</h3>
                            </div>
                            <div class="bg-blue-600/20 border border-blue-500/30 text-blue-400 px-3 py-1 rounded font-bold text-sm">
                                ${player.ovr} OVR
                            </div>
                        </div>
                        <div class="flex justify-between items-center pt-2 border-t border-slate-800">
                            <span class="text-yellow-400 font-bold">🪙 ${player.price.toLocaleString()}</span>
                            <button onclick="buyPlayer(${player.id})" class="bg-blue-600 hover:bg-blue-500 text-white font-bold px-4 py-1.5 rounded transition text-sm shadow">Sign Player</button>
                        </div>
                    </div>
                `;
            });
        }

        function buyPlayer(id) {
            const playerIndex = gameState.market.findIndex(p => p.id === id);
            if (playerIndex === -1) return;
            const player = gameState.market[playerIndex];

            if (gameState.coins >= player.price) {
                gameState.coins -= player.price;
                gameState.squad.push(player);
                gameState.market.splice(playerIndex, 1);
                saveGame();
            } else {
                alert("Not enough coins to sign this player!");
            }
        }

        function refreshMarket() {
            const pool = [
                { name: "Kylian Mbappé", pos: "FWD", ovr: 92, price: 1300 },
                { name: "Eden Hazard", pos: "FWD", ovr: 91, price: 1150 },
                { name: "Marcelo", pos: "DEF", ovr: 88, price: 800 },
                { name: "Casemiro", pos: "MID", ovr: 88, price: 850 },
                { name: "Thibaut Courtois", pos: "GK", ovr: 89, price: 950 },
                { name: "Antoine Griezmann", pos: "FWD", ovr: 89, price: 900 }
            ];
            gameState.market = pool.sort(() => 0.5 - Math.random()).slice(0, 4);
            saveGame();
        }

        // Match Engine Simulation
        let currentMatchReward = 100;
        let matchInterval = null;

        function startMatch(opponentName, reward) {
            currentMatchReward = reward;
            document.getElementById('matchSetup').classList.add('hidden');
            document.getElementById('matchLive').classList.remove('hidden');
            document.getElementById('matchTeamAway').innerText = opponentName;
            document.getElementById('scoreHome').innerText = '0';
            document.getElementById('scoreAway').innerText = '0';
            document.getElementById('matchMinute').innerText = '0\'';
            document.getElementById('matchResultActions').classList.add('hidden');
            
            const log = document.getElementById('commentaryLog');
            log.innerHTML = `<p class="text-slate-400">Referee blows the whistle! Match is underway against ${opponentName}...</p>`;

            let minute = 0;
            let homeGoals = 0;
            let awayGoals = 0;

            if (matchInterval) clearInterval(matchInterval);

            matchInterval = setInterval(() => {
                minute += Math.floor(Math.random() * 8) + 5;
                if (minute >= 90) {
                    minute = 90;
                    clearInterval(matchInterval);
                    finishMatchSimulation(homeGoals, awayGoals);
                }
                document.getElementById('matchMinute').innerText = minute + "'";

                // Match events simulation
                const eventRoll = Math.random();
                if (eventRoll < 0.35) {
                    homeGoals++;
                    document.getElementById('scoreHome').innerText = homeGoals;
                    log.innerHTML += `<p class="text-green-400 font-semibold">${minute}' - GOAL! Clinical finish by Dream FC!</p>`;
                } else if (eventRoll > 0.70 && awayGoals < homeGoals) {
                    awayGoals++;
                    document.getElementById('scoreAway').innerText = awayGoals;
                    log.innerHTML += `<p class="text-red-400 font-semibold">${minute}' - Goal for ${opponentName}...</p>`;
                } else {
                    const comments = [
                        `${minute}' - Midfield battle as both teams fight for possession.`,
                        `${minute}' - Great through ball, but intercepted by the defense.`,
                        `${minute}' - Powerful shot flies just wide of the goal!`,
                        `${minute}' - Excellent sliding tackle to break up the counter-attack.`
                    ];
                    log.innerHTML += `<p class="text-slate-300">${comments[Math.floor(Math.random() * comments.length)]}</p>`;
                }
                log.scrollTop = log.scrollHeight;
            }, 1000);
        }

        function finishMatchSimulation(home, away) {
            const log = document.getElementById('commentaryLog');
            if (home > away) {
                gameState.stats.wins++;
                gameState.coins += currentMatchReward;
                log.innerHTML += `<p class="text-green-400 font-bold mt-2">FULL TIME: Victory! You earned 🪙 +${currentMatchReward} coins.</p>`;
            } else if (home === away) {
                gameState.stats.draws++;
                gameState.coins += Math.floor(currentMatchReward / 3);
                log.innerHTML += `<p class="text-yellow-400 font-bold mt-2">FULL TIME: Draw. Points shared.</p>`;
            } else {
                gameState.stats.losses++;
                log.innerHTML += `<p class="text-red-400 font-bold mt-2">FULL TIME: Defeat. Dust yourselves off and try again.</p>`;
            }
            saveGame();
            document.getElementById('matchResultActions').classList.remove('hidden');
        }

        function endMatch() {
            document.getElementById('matchLive').classList.add('hidden');
            document.getElementById('matchSetup').classList.remove('hidden');
            switchTab('hub');
        }

        function upgradeStadium() {
            if (gameState.coins >= 500) {
                gameState.coins -= 500;
                gameState.stadiumCapacity += 5000;
                gameState.stadiumName = gameState.stadiumCapacity >= 30000 ? "Grand Arena" : "Elite Stadium";
                saveGame();
            } else {
                alert("You need 500 coins to upgrade your stadium!");
            }
        }

        function setTeamColor(colorName, className) {
            gameState.primaryColor = className;
            saveGame();
        }

        // Initialize App on Load
        updateUI();
    </script>
